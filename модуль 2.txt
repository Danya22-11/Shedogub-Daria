SET search_path TO pitanie, public;
SET lc_time = 'ru_RU.UTF-8';

-- Задание Б1. Длительность закрытых простоев (Топ-10)
SELECT downtime_id, line_id, start_time, end_time,
       (end_time - start_time)::text AS duration_interval,
       ROUND(EXTRACT(EPOCH FROM end_time - start_time) / 3600, 2) AS duration_hours
FROM equipment_downtime
WHERE end_time IS NOT NULL
ORDER BY duration_hours DESC
LIMIT 10;

-- Задание Б2. Незакрытые простои на дату среза 01.01.2026
SELECT downtime_id, line_id, start_time,
       (DATE '2026-01-01' - start_time::date) AS days_open,
       ROUND(EXTRACT(EPOCH FROM TIMESTAMP '2026-01-01 00:00' - start_time) / 3600, 2) AS hours_open
FROM equipment_downtime
WHERE end_time IS NULL
ORDER BY start_time ASC;

-- Задание Б4. Правильный расчет процента брака
SELECT batch_number, produced_qty, defect_qty,
       defect_qty / produced_qty AS formula_1,
       100 * defect_qty / produced_qty AS formula_2,
       ROUND(100.0 * defect_qty / produced_qty, 2) AS formula_3
FROM production_batches
WHERE batch_number IN ('П250602-3Д', 'П250602-3Н');

-- Задание Б5. Извлечение данных из номера партии
SELECT batch_number,
       to_date(SUBSTRING(batch_number FROM 2 FOR 6), 'YYMMDD') AS parsed_date,
       SPLIT_PART(batch_number, '-', 2) AS line_and_shift,
       RIGHT(batch_number, 1) AS shift_code
FROM production_batches
LIMIT 5;

-- Задание Б6. Топ-10 самых больших задержек поставок
SELECT poi.po_id, poi.material_id, po.expected_date, poi.received_date,
       (poi.received_date - po.expected_date) AS delay_days,
       GREATEST(poi.received_date - po.expected_date, 0) AS delay_positive
FROM purchase_order_items AS poi
JOIN purchase_orders AS po ON po.po_id = poi.po_id
WHERE poi.received_date IS NOT NULL
ORDER BY delay_days DESC
LIMIT 10;

-- Задание Б7. Возраст линий
SELECT line_id, line_name, launch_year,
       (2025 - launch_year) AS age_years,
       AGE(DATE '2025-12-31', make_date(launch_year, 1, 1)) AS age_exact,
       CASE 
           WHEN (2025 - launch_year) >= 15 THEN 'Кандидат на модернизацию'
           WHEN (2025 - launch_year) BETWEEN 8 AND 14 THEN 'Плановый контроль'
           ELSE 'В норме'
       END AS status
FROM production_lines
ORDER BY age_years DESC;

-- Задание Б8. Стоимость списаний (защита от деления на ноль)
SELECT m.material_name, mw.quantity, mw.amount,
       ROUND(mw.amount / NULLIF(mw.quantity, 0), 2) AS price_per_unit,
       ROUND(mw.amount, -3) AS amount_rounded
FROM material_writeoffs AS mw
JOIN raw_materials AS m ON m.material_id = mw.material_id
WHERE LOWER(mw.reason) LIKE '%истёк срок годности%'
ORDER BY mw.amount DESC
LIMIT 10;
