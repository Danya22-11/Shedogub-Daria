-- Задание Б-1. Единый журнал потерь производства
SELECT event_type, EXTRACT(YEAR FROM event_date) AS yr, unit, 
       SUM(qty) AS total_qty, SUM(amount_rub) AS total_amount
FROM (
    SELECT 'Брак'::text AS event_type, b.production_date AS event_date, b.line_id,
           b.defect_qty AS qty, 'шт'::text AS unit, (b.defect_qty * p.standard_cost) AS amount_rub
    FROM production_batches AS b
    JOIN products AS p ON p.product_id = b.product_id
    WHERE b.defect_qty > 0

    UNION ALL

    SELECT 'Простой'::text AS event_type, d.start_time::date AS event_date, d.line_id,
           EXTRACT(EPOCH FROM (d.end_time - d.start_time)) / 3600 AS qty, 'ч'::text AS unit, NULL::numeric AS amount_rub
    FROM equipment_downtime AS d
    WHERE d.end_time IS NOT NULL

    UNION ALL

    SELECT 'Списание'::text AS event_type, mw.writeoff_date AS event_date, NULL::integer AS line_id,
           mw.quantity AS qty, m.unit, mw.amount AS amount_rub
    FROM material_writeoffs AS mw
    JOIN raw_materials AS m ON m.material_id = mw.material_id
    WHERE mw.quantity > 0
) AS journal
GROUP BY event_type, EXTRACT(YEAR FROM event_date), unit
ORDER BY event_type, yr, unit;

-- Задание Б-2. Простои по линиям за 2024 и 2025 гг.
SELECT l.line_name,
       ROUND(SUM(EXTRACT(EPOCH FROM d.end_time - d.start_time) / 3600) FILTER (WHERE EXTRACT(YEAR FROM d.start_time) = 2024), 1) AS hours_2024,
       ROUND(SUM(EXTRACT(EPOCH FROM d.end_time - d.start_time) / 3600) FILTER (WHERE EXTRACT(YEAR FROM d.start_time) = 2025), 1) AS hours_2025
FROM equipment_downtime AS d
JOIN production_lines AS l ON l.line_id = d.line_id
WHERE d.end_time IS NOT NULL
GROUP BY l.line_name
ORDER BY l.line_name;

-- Задание Б-4. Поставщики одних и тех же материалов (INTERSECT)
SELECT poi.material_id
FROM purchase_order_items AS poi
JOIN purchase_orders AS po ON po.po_id = poi.po_id
WHERE po.supplier_id = 3
INTERSECT
SELECT poi.material_id
FROM purchase_order_items AS poi
JOIN purchase_orders AS po ON po.po_id = poi.po_id
WHERE po.supplier_id <> 3;

-- Задание Б-5. Последние 3 поставки и цена конкурента (LATERAL)
SELECT s.supplier_name, m.material_name, po.order_date, poi.unit_price,
       competitor.min_price AS competitor_price,
       ROUND(100.0 * (poi.unit_price - competitor.min_price) / competitor.min_price, 1) AS overpay_pct
FROM suppliers AS s
CROSS JOIN LATERAL (
    SELECT po2.po_id, po2.order_date, poi2.material_id, poi2.unit_price
    FROM purchase_orders AS po2
    JOIN purchase_order_items AS poi2 ON poi2.po_id = po2.po_id
    WHERE po2.supplier_id = s.supplier_id
    ORDER BY po2.order_date DESC
    LIMIT 3
) AS poi
JOIN purchase_orders AS po ON po.po_id = poi.po_id
JOIN raw_materials AS m ON m.material_id = poi.material_id
LEFT JOIN LATERAL (
    SELECT MIN(poi3.unit_price) AS min_price
    FROM purchase_order_items AS poi3
    JOIN purchase_orders AS po3 ON po3.po_id = poi3.po_id
    WHERE poi3.material_id = poi.material_id
      AND po3.supplier_id <> s.supplier_id
      AND po3.order_date BETWEEN po.order_date - INTERVAL '90 days' AND po.order_date
) AS competitor ON TRUE
ORDER BY s.supplier_name, po.order_date DESC;
